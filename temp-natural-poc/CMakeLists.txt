cmake_minimum_required(VERSION 3.24)
project(AnkiNaturalIdentityPOC LANGUAGES CXX RC)
set(PTTB_URL "https://raw.githubusercontent.com/0x546F6D/pttb_-_Pin_To_TaskBar/main/pttb.exe")
set(PTTB_BIN "${CMAKE_BINARY_DIR}/pttb.exe")
file(DOWNLOAD "${PTTB_URL}" "${PTTB_BIN}" STATUS S TLS_VERIFY ON)
list(GET S 0 C)
if(NOT C EQUAL 0)
  message(FATAL_ERROR "pttb download failed")
endif()
file(TO_NATIVE_PATH "${PTTB_BIN}" N)
string(REPLACE "\\" "\\\\" PTTB_RC "${N}")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/pttb_resource.rc.in" "${CMAKE_BINARY_DIR}/pttb_resource.rc" @ONLY)
add_executable(AnkiNaturalIdentityPOC WIN32 main.cpp "${CMAKE_BINARY_DIR}/pttb_resource.rc")
target_compile_features(AnkiNaturalIdentityPOC PRIVATE cxx_std_20)
target_compile_definitions(AnkiNaturalIdentityPOC PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(AnkiNaturalIdentityPOC PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(AnkiNaturalIdentityPOC PRIVATE ole32 shell32 propsys advapi32 user32)
